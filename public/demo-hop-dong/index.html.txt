// KHÓA CỨNG ID THƯ MỤC GOOGLE DRIVE CHO BẢN DEMO (Khách không thể sửa từ giao diện)
const LOCKED_DRIVE_FOLDER_ID = '10-d91Cf-_xYpx54B0uR-Fzvc48Jcwl8-';

// Hỗ trợ vừa chạy trực tiếp bằng HtmlService vừa làm REST API cho GitHub Pages
function doGet(e) {
  setupDatabase();
  if (e && e.parameter && e.parameter.action === 'getInitialData') {
    return ContentService.createTextOutput(JSON.stringify({ ok: true, data: getInitialData() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
  return HtmlService.createHtmlOutputFromFile('Index')
    .setTitle('Webapp Quản Lý Tài Liệu, Văn Bản (v1.0)')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL)
    .addMetaTag('viewport', 'width=device-width, initial-scale=1');
}

function doPost(e) {
  try {
    const body = JSON.parse(e.postData.contents || '{}');
    if (body.action === 'syncState') {
      return ContentService.createTextOutput(JSON.stringify(syncFullStateToSheets(body.state)))
        .setMimeType(ContentService.MimeType.JSON);
    }
    if (body.action === 'uploadFile') {
      body.payload.folderId = LOCKED_DRIVE_FOLDER_ID;
      return ContentService.createTextOutput(JSON.stringify(uploadFileToDrive(body.payload)))
        .setMimeType(ContentService.MimeType.JSON);
    }
    return ContentService.createTextOutput(JSON.stringify({ ok: false, message: 'Unknown action' }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ ok: false, message: err.message }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function verifyDriveFolder(folderInput) {
  try {
    const idMatch = String(folderInput || '').match(/[-\w]{25,}/);
    const folderId = idMatch ? idMatch[0] : String(folderInput || '').trim();
    const folder = DriveApp.getFolderById(folderId);
    return { ok: true, id: folderId, name: folder.getName() };
  } catch (err) {
    return { ok: false, message: 'Không tìm thấy thư mục hoặc chưa cấp quyền: ' + err.message };
  }
}

function uploadFileToDrive(payload) {
  try {
    const idMatch = String(payload.folderId || '').match(/[-\w]{25,}/);
    const folderId = idMatch ? idMatch[0] : String(payload.folderId || '').trim();
    const folder = folderId ? DriveApp.getFolderById(folderId) : DriveApp.getRootFolder();
    const bytes = Utilities.base64Decode(payload.base64Data);
    const blob = Utilities.newBlob(bytes, payload.mimeType || 'application/octet-stream', payload.fileName);
    const file = folder.createFile(blob);
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    return {
      ok: true,
      fileId: file.getId(),
      fileName: file.getName(),
      previewUrl: 'https://drive.google.com/file/d/' + file.getId() + '/preview',
      viewUrl: file.getUrl()
    };
  } catch (err) {
    return { ok: false, message: err.message };
  }

    function loadState() {
      try {
        const raw = localStorage.getItem(STORAGE_KEY);
        if (raw) {
          const parsed = JSON.parse(raw);
          const merged = Object.assign({}, DEFAULT_STATE, parsed);
          merged.driveConfig = Object.assign({}, LOCKED_DRIVE_CONFIG);
          // Ensure every user object has permissions object
          if (Array.isArray(merged.users)) {
            merged.users.forEach(u => {
              if (!u.permissions) {
                u.permissions = u.role === 'ADMIN'
                  ? { canView: true, canCreate: true, canEdit: true, canDelete: true }
                  : { canView: true, canCreate: true, canEdit: false, canDelete: false };
              }
            });
          }
          return merged;
        }
      } catch (e) {
        // Fallback to in-memory state if localStorage is blocked
      }
      return JSON.parse(JSON.stringify(DEFAULT_STATE));
    }

    function saveState() {
      state.driveConfig = Object.assign({}, LOCKED_DRIVE_CONFIG);
      try {
        localStorage.setItem(STORAGE_KEY, JSON.stringify(state));
      } catch (e) {
        // Ignore storage error in restricted iframe
      }
      if (IS_GAS_ENV) {
        try {
          google.script.run.syncFullStateToSheets(state);
        } catch (err) {}
      } else if (state.gasApiUrl && state.gasApiUrl.startsWith('https://script.google.com/')) {
        fetch(state.gasApiUrl, {
          method: 'POST',
          headers: { 'Content-Type': 'text/plain;charset=utf-8' },
          body: JSON.stringify({ action: 'syncState', state: state })
        }).catch(() => {});
      }
    }

    // Check if document is expiring within 30 days or already expired (Ref date: 2026-10-28)
    function getExpiryStatus(doc) {
      if (!doc.expiryDate) return { code: 'NORMAL', label: doc.status || 'Đang hiệu lực', badgeClass: 'bg-emerald-50 text-emerald-700 border-emerald-200' };
      const today = new Date('2026-10-28T00:00:00');
      const exp = new Date(doc.expiryDate + 'T00:00:00');
      const diffDays = Math.ceil((exp - today) / (1000 * 60 * 60 * 24));

      if (doc.status === 'Kết thúc' || doc.status === 'Đã thanh lý') {
        return { code: 'CLOSED', label: doc.status, badgeClass: 'bg-slate-100 text-slate-600 border-slate-200' };
      }
      if (diffDays < 0) {
        return { code: 'EXPIRED', label: `Quá hạn ${Math.abs(diffDays)} ngày`, badgeClass: 'bg-rose-50 text-rose-700 border-rose-200' };
      }
      if (diffDays <= 30) {
        return { code: 'EXPIRING', label: `Còn ${diffDays} ngày`, badgeClass: 'bg-amber-50 text-amber-700 border-amber-200' };
      }
      return { code: 'ACTIVE', label: 'Đang hiệu lực', badgeClass: 'bg-emerald-50 text-emerald-700 border-emerald-200' };
    }

    function detectFileTypeFromName(fileName) {
      const lower = (fileName || '').toLowerCase();
      if (lower.endsWith('.pdf')) return 'PDF';
      if (lower.endsWith('.doc') || lower.endsWith('.docx')) return 'WORD';
      if (lower.endsWith('.xls') || lower.endsWith('.xlsx') || lower.endsWith('.csv')) return 'EXCEL';
      if (lower.endsWith('.png') || lower.endsWith('.jpg') || lower.endsWith('.jpeg') || lower.endsWith('.webp')) return 'IMAGE';
      return 'PDF';
    }

        case 'EXCEL': return { icon: 'fa-file-excel', colorClass: 'text-emerald-600 bg-emerald-50 border-emerald-200', label: 'Excel' };
        case 'IMAGE': return { icon: 'fa-file-image', colorClass: 'text-purple-600 bg-purple-50 border-purple-200', label: 'Hình ảnh' };
        default: return { icon: 'fa-file-lines', colorClass: 'text-slate-600 bg-slate-100 border-slate-200', label: 'Khác' };
      }
    }

    function getFilteredDocuments() {
      const visibleDocs = getVisibleDocumentsForCurrentUser();
      const q = (document.getElementById('filterSearchInput').value || '').trim().toLowerCase();
      const dept = document.getElementById('filterDepartment').value;
      const proj = document.getElementById('filterProject').value;
      const vendor = document.getElementById('filterVendor').value;
      const owner = document.getElementById('filterOwner').value;
      const dateFrom = document.getElementById('filterDateFrom').value;
      const dateTo = document.getElementById('filterDateTo').value;

      return visibleDocs.filter(doc => {
        if (state.selectedCategoryId !== 'ALL' && doc.categoryId !== state.selectedCategoryId) return false;
        if (state.selectedFileType !== 'ALL' && doc.fileType !== state.selectedFileType) return false;
        if (state.onlyExpiring) {
          const st = getExpiryStatus(doc);
          if (st.code !== 'EXPIRED' && st.code !== 'EXPIRING') return false;
        }
        if (dept !== 'ALL' && doc.department !== dept) return false;
        if (proj !== 'ALL' && doc.project !== proj) return false;
        if (vendor !== 'ALL' && doc.vendor !== vendor) return false;
        if (owner !== 'ALL' && doc.owner !== owner) return false;
        if (dateFrom && doc.issueDate < dateFrom) return false;
        if (dateTo && doc.issueDate > dateTo) return false;
        if (q) {
          const hay = `${doc.code} ${doc.title} ${doc.description} ${doc.vendor} ${doc.project}`.toLowerCase();
          if (!hay.includes(q)) return false;
        }
        return true;
      });
    }

    function renderDocumentList() {
      const filtered = getFilteredDocuments();
      const user = getCurrentUserObj();
      const visibleCats = getVisibleCategoriesForCurrentUser();

      // Update Category Header & Total Contract Value
      const activeCat = state.categories.find(c => c.id === state.selectedCategoryId);
      const badgeEl = document.getElementById('currentCategoryBadge');
      const titleEl = document.getElementById('currentCategoryTitle');

      if (activeCat) {
        badgeEl.style.backgroundColor = activeCat.color;
        badgeEl.innerHTML = `<i class="fa-solid ${activeCat.icon}"></i>`;
        titleEl.textContent = activeCat.name;
      } else {
        badgeEl.style.backgroundColor = '#059669';
        badgeEl.innerHTML = `<i class="fa-solid fa-layer-group"></i>`;
        titleEl.textContent = 'Tổng tất cả các hồ sơ';
      }

      document.getElementById('currentFilterSummary').textContent =
        `Hiển thị ${filtered.length} hồ sơ • Tài khoản: ${user.fullName} (Được xem ${visibleCats.length}/${state.categories.length} danh mục)`;

      const totalVal = filtered.reduce((acc, d) => acc + (Number(d.value) || 0), 0);
      document.getElementById('totalContractValueDisplay').textContent = totalVal.toLocaleString('vi-VN') + ' ₫';

      const tbody = document.getElementById('documentTableBody');
      if (filtered.length === 0) {
        tbody.innerHTML = `
          <tr>
            <td colspan="8" class="py-12 text-center text-slate-400">
              <div class="w-12 h-12 rounded-2xl bg-slate-100 text-slate-400 flex items-center justify-center mx-auto mb-3 text-xl">
                <i class="fa-regular fa-folder-open"></i>
              </div>
              <p class="text-sm font-semibold text-slate-600">Không tìm thấy hồ sơ, tài liệu nào phù hợp</p>
              <p class="text-xs text-slate-400 mt-1">Hãy thử đặt lại bộ lọc hoặc nhấn "Thêm mới hồ sơ" để tải tệp lên.</p>
            </td>
          </tr>
        `;
        return;
      }

      if (state.groupBy === 'NONE') {
        tbody.innerHTML = filtered.map(doc => renderDocumentRow(doc)).join('');
      } else {
        // Grouped rendering (As in Video 01:58)
        const groups = {};
        filtered.forEach(doc => {
          let keyLabel = doc[state.groupBy] || 'Chưa phân loại';
          if (state.groupBy === 'categoryId') {
            const c = state.categories.find(x => x.id === doc.categoryId);
            keyLabel = c ? c.name : 'Danh mục khác';
          }
          if (!groups[keyLabel]) groups[keyLabel] = [];
          groups[keyLabel].push(doc);
        });

        let html = '';
        Object.keys(groups).forEach(groupName => {
          const items = groups[groupName];
          const groupSum = items.reduce((s, x) => s + (Number(x.value) || 0), 0);
          html += `
            <tr class="bg-slate-100/90 border-y border-slate-200">
              <td colspan="8" class="py-2.5 px-4 text-xs font-bold text-slate-800">
                <div class="flex items-center justify-between">
                  <span><i class="fa-solid fa-folder-tree text-emerald-600 mr-2"></i>${groupName} <span class="ml-1.5 px-2 py-0.5 rounded-full bg-emerald-100 text-emerald-800 text-[11px]">${items.length} hồ sơ</span></span>
                  <span class="text-slate-500 font-semibold">Tổng nhóm: <strong class="text-emerald-700">${formatCurrency(groupSum)}</strong></span>
                </div>
              </td>
            </tr>
          `;
          html += items.map(doc => renderDocumentRow(doc)).join('');
        });
        tbody.innerHTML = html;
      }

      // Attach Row Action Listeners
      tbody.querySelectorAll('button[data-preview-id]').forEach(btn => {
        btn.addEventListener('click', () => openPreviewModal(btn.getAttribute('data-preview-id')));
      });
      tbody.querySelectorAll('button[data-edit-id]').forEach(btn => {
        btn.addEventListener('click', () => openDocModal(btn.getAttribute('data-edit-id')));
      });
      tbody.querySelectorAll('button[data-delete-id]').forEach(btn => {
        btn.addEventListener('click', () => deleteDocument(btn.getAttribute('data-delete-id')));
      });
    }

    function bindPermissionCardEvents(containerEl) {
      if (!containerEl) return;

      containerEl.querySelectorAll('input[data-card-perm-user]').forEach(chk => {
        chk.addEventListener('change', () => {
          const uname = chk.getAttribute('data-card-perm-user');
          const catId = chk.getAttribute('data-card-perm-cat');
          const u = state.users.find(x => x.username === uname);
          if (!u || u.role === 'ADMIN') return;
          if (chk.checked) {
            if (!u.allowedCategories.includes(catId)) u.allowedCategories.push(catId);
          } else {
            u.allowedCategories = u.allowedCategories.filter(id => id !== catId);
          }
          saveState();
        });
      });

      containerEl.querySelectorAll('input[data-action-user]').forEach(chk => {
        chk.addEventListener('change', () => {
          const uname = chk.getAttribute('data-action-user');
          const actionKey = chk.getAttribute('data-action-key');
          const u = state.users.find(x => x.username === uname);
          if (!u || u.role === 'ADMIN') return;
          if (!u.permissions) u.permissions = { canView: true, canCreate: true, canEdit: false, canDelete: false };
          u.permissions[actionKey] = chk.checked;
          saveState();
        });
      });

      containerEl.querySelectorAll('input[data-pwd-user]').forEach(inp => {
        inp.addEventListener('change', () => {
          const uname = inp.getAttribute('data-pwd-user');
          const u = state.users.find(x => x.username === uname);
          if (!u) return;
          u.password = inp.value.trim() || '123456';
          saveState();
        });
      });

      containerEl.querySelectorAll('button[data-select-all-cats]').forEach(btn => {
        btn.addEventListener('click', () => {
          const uname = btn.getAttribute('data-select-all-cats');
          const u = state.users.find(x => x.username === uname);
          if (!u || u.role === 'ADMIN') return;
          const allIds = state.categories.map(c => c.id);
          if (u.allowedCategories.length === allIds.length) {
            u.allowedCategories = [];
          } else {
            u.allowedCategories = allIds.slice();
          }
          saveState();
          renderAll();
          showToast(`Đã cập nhật nhanh danh mục cho tài khoản ${u.username}!`);
        });
      });

      containerEl.querySelectorAll('button[data-del-user]').forEach(btn => {
        btn.addEventListener('click', () => {
          const uname = btn.getAttribute('data-del-user');
          if (uname === 'admin') return;
          state.users = state.users.filter(x => x.username !== uname);
          if (state.currentUser === uname) state.currentUser = 'admin';
          saveState();
          renderAll();
          showToast(`Đã xóa tài khoản @${uname}!`);
        });
      });

      containerEl.querySelectorAll('button[data-save-user-perm]').forEach(btn => {
        btn.addEventListener('click', () => {
          const uname = btn.getAttribute('data-save-user-perm');
          saveState();
          renderAll();
          showToast(`Đã lưu phân quyền cho tài khoản "${uname}"!`);
        });
      });
    }

    // Export filtered documents to CSV (Excel compatible UTF-8 BOM)
    function exportFilteredToExcelCsv() {
      const rows = getFilteredDocuments();
      if (rows.length === 0) {
        showToast('Không có dữ liệu hồ sơ để xuất!', 'error');
        return;
      }
      const headers = ['Số Hồ Sơ', 'Tên Tài Liệu', 'Danh Mục', 'Phòng Ban', 'Dự Án', 'Nhà Cung Cấp', 'Ngày Ban Hành', 'Ngày Kết Thúc', 'Giá Trị HĐ', 'Tình Trạng', 'Tên Tệp Drive', 'Loại Tệp', 'Người Phụ Trách'];
      const csvLines = [headers.join(',')];

      rows.forEach(d => {
        const cat = state.categories.find(c => c.id === d.categoryId);
        const line = [
          d.code,
          d.title,
          cat ? cat.name : '',
          d.department,
          d.project,
          d.vendor,
          d.issueDate,
          d.expiryDate,
          d.value || 0,
          d.status,
          d.fileName,
          d.fileType,
          d.owner
        ].map(val => `"${String(val || '').replace(/"/g, '""')}"`);
        csvLines.push(line.join(','));
      });

      const blob = new Blob(['\uFEFF' + csvLines.join('\n')], { type: 'text/csv;charset=utf-8;' });
      const url = URL.createObjectURL(blob);
      const a = document.createElement('a');
      a.href = url;
      a.download = `Danh_Sach_Ho_So_Tai_Lieu_${new Date().toISOString().slice(0, 10)}.csv`;
      document.body.appendChild(a);
      a.click();
      a.remove();
      URL.revokeObjectURL(url);
      showToast(`Đã xuất ${rows.length} hồ sơ ra file Excel (.csv) thành công!`);
    }

            fileType: fileType,
            owner: existing ? existing.owner : getCurrentUserObj().fullName,
            previewDataUrl: dataUrl || (existing ? existing.previewDataUrl : ''),
            drivePreviewUrl: drivePreviewUrl || (existing ? (existing.drivePreviewUrl || '') : '')
          };

          if (existing) {
            const idx = state.documents.findIndex(d => d.id === editId);
            state.documents[idx] = docPayload;
            showToast(`Đã cập nhật hồ sơ "${docPayload.code}" thành công!`);
          } else {
            state.documents.unshift(docPayload);
            showToast(`Đã lưu hồ sơ "${docPayload.code}" vào thư mục Drive "${state.driveConfig.folderName}"!`);
          }

          saveState();
          closeDocModal();
          renderAll();
        };

          showToast('Đang tải tệp lên Google Drive...');
          const reader = new FileReader();
          reader.onload = (ev) => {
            const base64Data = String(ev.target.result).split(',')[1] || '';
            google.script.run.withSuccessHandler((res) => {
              if (res && res.ok) {
                finalizeSave('', res.previewUrl);
              } else {
                showToast('Lỗi tải tệp Drive: ' + (res ? res.message : ''), 'error');
                finalizeSave('', '');
              }
            }).uploadFileToDrive({
              folderId: state.driveConfig.folderId,
              fileName: uploadedFile.name,
              mimeType: uploadedFile.type,
              base64Data: base64Data
            });
          };
          reader.readAsDataURL(uploadedFile);
        } else if (uploadedFile && uploadedFile.type.startsWith('image/')) {
          const reader = new FileReader();
          reader.onload = (ev) => finalizeSave(ev.target.result, '');
          reader.readAsDataURL(uploadedFile);
        } else {
          finalizeSave('', '');
        }
      });
